    ```typescript
    // frontend/src/composables/useTotTree.ts

    import { shallowRef, triggerRef } from 'vue'
    import type { TotNodeEvent, TotPruneEvent } from '@/types/agent'

    /**
     * 前端树节点，从 TotNodeEvent 流重建。
     */
    export interface TotTreeNode {
      nodeId: number
      level: number
      thought: string
      score: number
      parentId: number | null
      pruned: boolean
      children: TotTreeNode[]
    }
    ```

    实现 `useTotTree()` composable，返回 `{ nodeMap, rootNodes, addNode, markPruned, getLeafNodes, getOptimalPath, getNodesByLevel, clear }`：

    - `nodeMap = shallowRef<Map<number, TotTreeNode>>(new Map())` -- 全部节点索引
    - `rootNodes = shallowRef<TotTreeNode[]>([])` -- 根节点列表（parentId=null）
    - `addNode(ev: TotNodeEvent)`: 构造 TotTreeNode（pruned=false, children=[]），放入 nodeMap；若 parentId 非空则挂到父节点的 children（`parent.children = [...parent.children, node]`），否则加入 rootNodes。**每次变更后调用 `triggerRef(nodeMap)`** -- shallowRef 内部 mutation 需显式触发响应式（getNodesByLevel/getOptimalPath 在 render 中读取 nodeMap.value，triggerRef 确保剪枝/追加触发重渲染）。
    - `markPruned(ev: TotPruneEvent)`: 遍历 ev.prunedNodeIds，将 nodeMap 中对应节点的 pruned 置 true。变更后调用 `triggerRef(nodeMap)`。
    - `getLeafNodes()`: nodeMap 值中 children.length === 0 的节点。
    - `getOptimalPath(): TotTreeNode[]`: 叶子节点中选最高分（`reduce((a, b) => a.score >= b.score ? a : b)`），沿 parentId 回溯到根，unshift 顺序为根 -> 叶子。
    - `getNodesByLevel(level: number)`: nodeMap 值中 level 匹配的节点。
    - `clear()`: 重置 nodeMap 和 rootNodes 为空。

    带 JSDoc 注释，风格与现有 composables 一致（中文注释 + Phase 标注）。

    **渲染模式判别（per UI-SPEC §ReasoningPanel）：**

    在 `<script setup>` 中添加 computed：
    ```typescript
    // Phase 8: ToT 树状渲染模式 -- 选中 tot 且树有根节点时启用
    const isTotMode = computed(() =>
      props.selectedPattern === 'tot'
      && !!props.totTree
      && props.totTree.rootNodes.value.length > 0
    )

    // Phase 8: 层级列表 [-1, 0, 1, ..., maxLevel]（-1 = 原始问题根节点）
    const totLevels = computed(() => {
      if (!isTotMode.value || !props.totTree) return []
      const levels = [-1]
      for (const n of props.totTree.nodeMap.value.values()) {
        if (n.level > levels[levels.length - 1]) levels.push(n.level)
      }
      return levels
    })

    // Phase 8: 判断节点是否在最优路径上
    function isOptimalPath(nodeId: number): boolean {
      return props.totTree?.getOptimalPath().some(n => n.nodeId === nodeId) ?? false
    }

    // Phase 8: 评分徽章三段配色（>=7 金 / 4-6 琥珀 / <=3 红）
    function scoreClass(score: number): string {
      if (score >= 7) return 'high'
      if (score >= 4) return 'mid'
      return 'low'
    }
    ```
